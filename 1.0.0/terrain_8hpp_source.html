294 template <typename F>
295 void for_each_brush_cell(int cx, int cz, int brush_radius, F&& fn) {
296 for (int r = cz - brush_radius; r <= cz + brush_radius; ++r) {
297 for (int c = cx - brush_radius; c <= cx + brush_radius; ++c) {
298 if (!this->in_bounds(c, r)) continue;
299 int dx = c - cx, dz = r - cz;
300 float dist = std::sqrt(static_cast<float>(dx * dx + dz * dz));
301 if (dist <= brush_radius) fn(c, r, dist);
302 }
303 }
304 }
