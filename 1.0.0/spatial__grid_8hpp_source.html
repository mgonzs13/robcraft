97 void traverse_ray(const Ray& ray, double max_range, F&& callback) const {
98 double ox = ray.origin.x;
99 double oz = ray.origin.z;
100 double dx = ray.direction.x;
101 double dz = ray.direction.z;
102
103 int cx = this->cell_x(ox);
104 int cz = this->cell_z(oz);
105
106 if (cx < 0 || cx >= this->cols_ || cz < 0 || cz >= this->rows_) return;
107
108 int step_x = (dx > 0) ? 1 : (dx < 0) ? -1 : 0;
109 int step_z = (dz > 0) ? 1 : (dz < 0) ? -1 : 0;
110
111 double t_delta_x = (std::abs(dx) > 1e-12) ? std::abs(this->cell_size_ / dx)
112 : std::numeric_limits<double>::infinity();
113 double t_delta_z = (std::abs(dz) > 1e-12) ? std::abs(this->cell_size_ / dz)
114 : std::numeric_limits<double>::infinity();
115
116 double next_x = (dx > 0) ? (this->origin_.x + (cx + 1) * this->cell_size_)
117 : (this->origin_.x + cx * this->cell_size_);
118 double t_max_x =
119 (std::abs(dx) > 1e-12) ? (next_x - ox) / dx : std::numeric_limits<double>::infinity();
120
121 double next_z = (dz > 0) ? (this->origin_.y + (cz + 1) * this->cell_size_)
122 : (this->origin_.y + cz * this->cell_size_);
123 double t_max_z =
124 (std::abs(dz) > 1e-12) ? (next_z - oz) / dz : std::numeric_limits<double>::infinity();
125
126 double t = 0.0;
127
128 while (t < max_range) {
129 if (callback(cz, cx)) break;
130
131 if (t_max_x < t_max_z) {
132 t = t_max_x;
133 if (t > max_range) break;
134 cx += step_x;
135 if (cx < 0 || cx >= this->cols_) break;
136 t_max_x += t_delta_x;
137 } else {
138 t = t_max_z;
139 if (t > max_range) break;
140 cz += step_z;
141 if (cz < 0 || cz >= this->rows_) break;
142 t_max_z += t_delta_z;
143 }
144 }
145 }
